    let marca = 'Audi';

    let stock = 349;

    const modelos = [

        'Q8',

    ];


    /*

    const content = document.querySelector('#content');

    if (!content)

    content.innerHTML = `

        <br>
        <p> Marca: <span> ${marca} </span> </p>
        <p> Velocidad: <span> ${velocidad} </span> </p>
        <p> Stock: <span> ${stock} </span> </p>

        <h3> Modelos disponibles: </h3>
        <ul>
